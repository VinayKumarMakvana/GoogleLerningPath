import { CareerService } from '../career/career.service';
import { PlannerService } from '../planner/planner.service';
import { ProgressService } from '../progress/progress.service';
import { PriorityEngineService } from '../mentor/priority-engine.service';
import { PrismaClient } from '@prisma/client';

const prisma = new PrismaClient();

export class DashboardService {
  private careerService: CareerService;
  private plannerService: PlannerService;
  private progressService: ProgressService;
  private priorityEngine: PriorityEngineService;

  constructor() {
    this.careerService = new CareerService();
    this.plannerService = new PlannerService();
    this.progressService = new ProgressService();
    this.priorityEngine = new PriorityEngineService();
  }

  async getDashboard(userId: string) {
    const today = new Date();
    today.setHours(0, 0, 0, 0);

    const [
      careerDashboard,
      todayPlan,
      progressIntelligence,
      topicsProgress,
      rankedActions,
      recentPracticeAttempts
    ] = await Promise.all([
      this.careerService.getCareerDashboard(userId),
      this.plannerService.getTodayPlan(userId),
      this.progressService.getIntelligence(userId),
      this.progressService.getTopicsProgress(userId),
      this.priorityEngine.getRankedActions(userId),
      prisma.practiceAttempt.findMany({
        where: { userId },
        orderBy: { updatedAt: 'desc' },
        take: 30
      })
    ]);

    // Calculate a derived streak from practice attempts + progress updates
    let currentStreak = 0;
    // For a real streak, we would look at consecutive days in the database where an action occurred.
    // We can do a basic calculation here using recent practice attempts.
    if (recentPracticeAttempts.length > 0) {
      const dates = new Set(recentPracticeAttempts.map(a => {
        const d = new Date(a.updatedAt);
        d.setHours(0, 0, 0, 0);
        return d.getTime();
      }));
      
      let d = new Date();
      d.setHours(0, 0, 0, 0);
      
      // If the user practiced today or yesterday, streak continues
      if (dates.has(d.getTime())) {
        currentStreak++;
        d.setDate(d.getDate() - 1);
        while (dates.has(d.getTime())) {
          currentStreak++;
          d.setDate(d.getDate() - 1);
        }
      } else {
        d.setDate(d.getDate() - 1);
        if (dates.has(d.getTime())) {
          currentStreak++;
          d.setDate(d.getDate() - 1);
          while (dates.has(d.getTime())) {
            currentStreak++;
            d.setDate(d.getDate() - 1);
          }
        }
      }
    }

    // Determine Q&As (mocking to 0 since there is no table)
    const qaCount = 0;

    return {
      profile: careerDashboard.profile,
      todayPlan,
      progress: progressIntelligence,
      topicsProgress,
      mentorRecommendations: rankedActions.slice(0, 3), // Top 3 priority actions
      careerReadiness: careerDashboard.careerReadiness,
      streak: currentStreak,
      qAndA: qaCount
    };
  }
}

import { DashboardService } from './dashboard.service';

const dashboardService = new DashboardService();

export class DashboardController {
  async getDashboard(req: Request, res: Response) {
    try {
      const userId = req.user!.id;
      const data = await dashboardService.getDashboard(userId);
      res.json(data);
    } catch (error: any) {
      console.error(error);
      res.status(500).json({ error: 'Failed to fetch dashboard aggregation' });
    }
  }
}

import { PrismaClient } from '@prisma/client';
import { DashboardService } from '../src/modules/dashboard/dashboard.service';

const prisma = new PrismaClient();
const dashboardService = new DashboardService();

async function run() {
  console.log('--- DASHBOARD INTEGRATION TEST ---');
  
  // 1. Get a user
  const user = await prisma.user.findFirst({
    include: { careerProfile: true, profile: true }
  });

  if (!user) {
    console.log('No user found to test with.');
    return;
  }

  console.log(`Testing with User: ${user.email} (ID: ${user.id})`);

  try {
    // 2. Fetch the dashboard data
    console.log('\nFetching dashboard aggregation...');
    const dashboard = await dashboardService.getDashboard(user.id);
    
    // 3. Verify elements
    console.log('\n--- VERIFICATION ---');
    console.log('1. Profile Data:', dashboard.profile ? '✅ YES' : '❌ NO');
    console.log('2. Today Plan:', dashboard.todayPlan ? `✅ YES (${dashboard.todayPlan.tasks?.length || 0} tasks)` : '❌ NO');
    
    console.log('3. Overall Progress:', dashboard.progress ? `✅ YES (${dashboard.progress.overall?.percentage}%)` : '❌ NO');
    console.log('4. Topics Progress:', dashboard.topicsProgress ? `✅ YES (${dashboard.topicsProgress.length} topics)` : '❌ NO');
    
    console.log('5. Mentor Recommendations:', dashboard.mentorRecommendations ? `✅ YES (${dashboard.mentorRecommendations.length} actions)` : '❌ NO');
    console.log('6. Career Readiness:', dashboard.careerReadiness ? '✅ YES' : '❌ NO');
    console.log('7. Derived Streak:', `✅ YES (${dashboard.streak} days)`);
    
    console.log('\n--- DASHBOARD INTEGRATION SUCCESSFUL ---');
    
  } catch (err) {
    console.error('\n❌ ERROR during dashboard integration test:', err);
  } finally {
    await prisma.$disconnect();
  }
}

run();
